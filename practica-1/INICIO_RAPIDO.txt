💾 ESTRUCTURA DE DATOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CSV tiene columnas:
  • Bitness: bs32 o bs64
  • Compilation: método de compilación
  • Entry_Size: tamaño de matriz (512, 1024, 2048, 4096)
  • Time_Output: tiempo en segundos
  • GFLOPS: gigaflops alcanzados
